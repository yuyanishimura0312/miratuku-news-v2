function build_schema_graph(page_meta):
  graph = {
    "@context": "https://schema.org",
    "@graph": [
      {
        "@type": "Organization",
        "@id": page_meta.org_url + "#org",
        "name": page_meta.org_name,
        "url": page_meta.org_url,
        "logo": page_meta.logo_url,
        "sameAs": page_meta.sameAs_list  # wikipedia, linkedin, etc
      },
      {
        "@type": "WebSite",
        "@id": page_meta.site_url + "#site",
        "url": page_meta.site_url,
        "name": page_meta.site_name,
        "publisher": {"@id": page_meta.org_url + "#org"}

        "@id": page_meta.url + "#article",
        "headline": page_meta.title,
        "datePublished": page_meta.published,
        "dateModified": page_meta.modified,
        "author": {"@type": "Person", "name": page_meta.author},

Mercator Crawl Frontier

Mercator クロールフロンティア crawling_indexing crawling 1999

Heydon-Najork (1999) の分散クローラ。Googlebot / Bingbot / Apache Nutch の祖。Priority queue + politeness + duplicate detection。

時間計算量 O(|pages| * fetch_latency) 空間計算量 O(|frontier| + |visited|)
擬似コード 
function Mercator_crawl(seed_urls, max_pages):
  frontier = PriorityQueue()
  for url in seed_urls:
    frontier.push(url, priority=compute_priority(url))
  visited = set()
  while not frontier.empty() and len(visited) < max_pages:
    url = frontier.pop()
    if url in visited or not robots_allow(url): continue
    visited.add(url)
    page = fetch(url)
    for link in extract_links(page):
      if link not in visited:
        frontier.push(link, priority=compute_priority(link))
  return visited

function generate_sitemap(urls, lastmod_map, output_path):
  with open(output_path, 'w') as f:
    f.write('<?xml version="1.0" encoding="UTF-8"?>\n')
    f.write('<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n')
    for url in urls:
      if not is_indexable(url): continue
      f.write('  <url>\n')
      f.write(f'    <loc>{xml_escape(url)}</loc>\n')
      f.write(f'    <lastmod>{lastmod_map[url]}</lastmod>\n')
      f.write(f'    <changefreq>{infer_changefreq(url)}</changefreq>\n')
      f.write(f'    <priority>{compute_priority(url):.1f}</priority>\n')
      f.write('  </url>\n')
    f.write('</urlset>\n')

function thompson_sampling(arms, n_rounds):
  # Each arm has Beta(alpha, beta) belief over success rate
  alpha = [1] * len(arms)
  beta  = [1] * len(arms)
  history = []
  for t in range(n_rounds):
    # Sample success rate from each arm's posterior
    samples = [random.betavariate(alpha[a], beta[a]) for a in range(len(arms))]
    # Pull arm with highest sample
    chosen = argmax(samples)
    reward = arms[chosen].pull()  # 0 or 1
    # Update posterior
    if reward == 1: alpha[chosen] += 1
    else: beta[chosen] += 1
    history.append((chosen, reward))
  return history

function haversine(lat1, lon1, lat2, lon2):
  R = 6371  # Earth radius in km
  phi1, phi2 = radians(lat1), radians(lat2)

  dlambda = radians(lon2 - lon1)

Levenshtein Edit Distance

Levenshtein (1965) 動的計画法による文字列編集距離。Spell correction / Fuzzy search の基礎。

$$D(i,j) = \min\begin{cases} D(i-1,j) + 1 \\ D(i,j-1) + 1 \\ D(i-1,j-1) + [s_i \neq t_j] \end{cases}$$
時間計算量 O(m * n) 空間計算量 O(min(m,n)) with row optimization
擬似コード 
function edit_distance(s1, s2):
  m, n = len(s1), len(s2)
  D = [[0]*(n+1) for _ in range(m+1)]
  for i in range(m+1): D[i][0] = i
  for j in range(n+1): D[0][j] = j
  for i in 1..m:
    for j in 1..n:
      cost = 0 if s1[i-1] == s2[j-1] else 1
      D[i][j] = min(
        D[i-1][j] + 1,      # deletion
        D[i][j-1] + 1,      # insertion
        D[i-1][j-1] + cost  # substitution
      )
  return D[m][n]

Gyöngyi-Garcia-Molina-Pedersen (2004) VLDB の信頼伝播ベース PageRank。Spam サイトを統計的に検出。

function TrustRank(graph, seed_trusted, d=0.85, max_iter=100):
  N = |graph.nodes|
  # Initialize: trust = uniform over trusted seeds, 0 elsewhere
  trust = {v: 1/|seed_trusted| if v in seed_trusted else 0 for v in graph.nodes}
  for iter in range(max_iter):
    trust_new = {v: 0 for v in graph.nodes}
    for v in graph.nodes:
      if v in seed_trusted:
        trust_new[v] = (1-d) * (1/|seed_trusted|)
      for u in graph.in_edges(v):
        trust_new[v] += d * trust[u] / |graph.out_edges(u)|
    trust = trust_new
  return trust  # high = trusted, low = candidate spam
