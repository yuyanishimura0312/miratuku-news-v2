// Canvas rAF loop with delta time (fps-independent)
let last=performance.now(),raf;
function frame(now){const dt=(now-last)/1000;last=now;
  ctx.clearRect(0,0,cv.width,cv.height);
  x+=vel*dt; ctx.fillRect(x,y,10,10);
  raf=requestAnimationFrame(frame);}

function tick(){ctx.clearRect(0,0,W,H);for(const p of ps){p.x+=p.vx;p.y+=p.vy;p.vy+=.05;p.life-=.01;ctx.globalAlpha=Math.max(0,p.life);ctx.fillRect(p.x,p.y,3,3)}requestAnimationFrame(tick)}tick();
